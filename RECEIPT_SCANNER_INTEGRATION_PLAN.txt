export async function processReceipt(
    imageBase64: string,
    model: string,
    keys: {...}
): Promise<ReceiptData> {
    // ... existing code to get apiKey and provider ...

    const prompt = `...`; // Updated prompt from Task 1.1

    console.log('[RECEIPT-SCANNER] Calling AI vision API...');
    const response = await generateVisionResponse(
        prompt,
        imageBase64,
        model,
        apiKey,
        provider
    );

    console.log('[RECEIPT-SCANNER] Raw AI response:', response);

    // Parse JSON response
    let data: ReceiptData;
    try {
        // Remove markdown code blocks if present
        let cleanResponse = response.trim();
        if (cleanResponse.startsWith('```')) {
            cleanResponse = cleanResponse.replace(/```json\n?/g, '').replace(/```\n?/g, '');
        }

        data = JSON.parse(cleanResponse);
    } catch (parseError) {
        console.error('[RECEIPT-SCANNER] JSON parse error:', parseError);
        throw new Error('Failed to parse AI response as JSON');
    }

    // Validate response structure
    if (!data.transaction || !data.items || !Array.isArray(data.items)) {
        console.error('[RECEIPT-SCANNER] Invalid response structure:', data);
        throw new Error('Invalid receipt data structure from AI');
    }

    // Validate required fields
    if (!data.transaction.store_name || !data.transaction.date || !data.transaction.total_amount) {
        throw new Error('Missing required transaction fields');
    }

    // Validate items
    if (data.items.length === 0) {
        console.warn('[RECEIPT-SCANNER] No items found, creating default item');
        // Create a single item with total amount
        data.items = [{
            name: 'Belanja (detail tidak tersedia)',
            quantity: 1,
            unit_price: data.transaction.total_amount,
            total_price: data.transaction.total_amount
        }];
    }

    // Validate total matches sum of items
    const itemsTotal = data.items.reduce((sum, item) => sum + item.total_price, 0);
    const difference = Math.abs(itemsTotal - data.transaction.total_amount);

    if (difference > 1) { // Allow 1 rupiah rounding difference
        console.warn('[RECEIPT-SCANNER] Total mismatch!', {
            transactionTotal: data.transaction.total_amount,
            itemsTotal,
            difference
        });

        // Auto-adjust: add/subtract difference to first item
        if (data.items.length > 0) {
            const adjustment = data.transaction.total_amount - itemsTotal;
            data.items[0].total_price += adjustment;
            data.items[0].notes = (data.items[0].notes || '') + ' (disesuaikan)';
            console.log('[RECEIPT-SCANNER] Adjusted first item by:', adjustment);
        }
    }

    console.log('[RECEIPT-SCANNER] Validated receipt data:', data);
    return data;
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import prisma from '@/lib/prisma';
import { processReceipt } from '@/lib/receipt-scanner';

export async function POST(request: NextRequest) {
    try {
        console.log('[SCAN-RECEIPT] Request received');

        const session = await getServerSession(authOptions);
        if (!session?.user?.id) {
            console.log('[SCAN-RECEIPT] Unauthorized - no session');
            return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
        }
        console.log('[SCAN-RECEIPT] User authenticated:', session.user.id);

        // Parse request (JSON or FormData)
        const contentType = request.headers.get('content-type') || '';
        let imageBase64: string;
        let model: string | null = null;
        let walletId: string | null = null;
        let categoryId: string | null = null;

        if (contentType.includes('application/json')) {
            const body = await request.json();
            imageBase64 = body.imageBase64;
            model = body.model || null;
            walletId = body.walletId || null;
            categoryId = body.categoryId || null;

            if (!imageBase64) {
                return NextResponse.json({ error: 'Image base64 is required' }, { status: 400 });
            }

            if (!imageBase64.startsWith('data:image')) {
                imageBase64 = `data:image/jpeg;base64,${imageBase64}`;
            }
        } else {
            const formData = await request.formData();
            const file = formData.get('file') as File;
            model = formData.get('model') as string | null;
            walletId = formData.get('walletId') as string | null;
            categoryId = formData.get('categoryId') as string | null;

            if (!file) {
                return NextResponse.json({ error: 'Image file is required' }, { status: 400 });
            }

            const bytes = await file.arrayBuffer();
            const buffer = Buffer.from(bytes);
            imageBase64 = `data:${file.type};base64,${buffer.toString('base64')}`;
        }

        // Fetch user settings for API keys
        const userSettings = await prisma.userSettings.findUnique({
            where: { userId: session.user.id }
        });

        if (!userSettings) {
            return NextResponse.json({
                error: 'Settings not found. Please configure your AI settings first.'
            }, { status: 404 });
        }

        const keys = {
            openaiApiKey: userSettings.openaiApiKey,
            geminiApiKey: userSettings.geminiApiKey,
            defaultApiKey: userSettings.apiKey,
            defaultProvider: userSettings.provider
        };

        // Process receipt with AI
        console.log('[SCAN-RECEIPT] Processing receipt...');
        const receiptData = await processReceipt(
            imageBase64,
            model || 'google/gemini-2.0-flash-exp:free',
            keys
        );

        // Get default wallet if not provided
        if (!walletId) {
            const defaultWallet = await prisma.wallet.findFirst({
                where: {
                    userId: session.user.id,
                    isDefault: true
                }
            });

            if (!defaultWallet) {
                // Get any wallet
                const anyWallet = await prisma.wallet.findFirst({
                    where: { userId: session.user.id }
                });

                if (!anyWallet) {
                    return NextResponse.json({
                        error: 'No wallet found. Please create a wallet first.'
                    }, { status: 400 });
                }

                walletId = anyWallet.id;
            } else {
                walletId = defaultWallet.id;
            }
        }

        // Get category ID if not provided
        if (!categoryId) {
            // Try to find "Belanja" category
            const category = await prisma.category.findFirst({
                where: {
                    userId: session.user.id,
                    name: receiptData.transaction.category || 'Belanja',
                    type: 'EXPENSE'
                }
            });

            if (category) {
                categoryId = category.id;
            }
        }

        // Create transaction with items in a single transaction
        const transaction = await prisma.transaction.create({
            data: {
                userId: session.user.id,
                walletId: walletId,
                categoryId: categoryId,
                description: receiptData.transaction.description,
                amount: receiptData.transaction.total_amount,
                date: new Date(receiptData.transaction.date),
                type: 'EXPENSE',
                // Create items inline
                items: {
                    create: receiptData.items.map((item, index) => ({
                        name: item.name,
                        quantity: item.quantity,
                        unitPrice: item.unit_price,
                        totalPrice: item.total_price,
                        notes: item.notes,
                        category: item.category,
                        order: index
                    }))
                }
            },
            include: {
                items: true,
                wallet: true,
                category: true
            }
        });

        console.log('[SCAN-RECEIPT] Success! Created transaction:', transaction.id);
        console.log('[SCAN-RECEIPT] Created items:', transaction.items.length);

        return NextResponse.json({
            success: true,
            transaction,
            message: `Receipt scanned! Created 1 transaction with ${transaction.items.length} items.`
        });

    } catch (error) {
        console.error('[SCAN-RECEIPT] Error:', error);
        return NextResponse.json(
            { error: error instanceof Error ? error.message : 'Failed to scan receipt' },
            { status: 500 }
        );
    }
}
```

```typescript
import { prisma } from '@/lib/prisma';

async function migrateReceiptTransactions() {
    console.log('Finding receipt transactions to migrate...');

    // Find transactions that look like receipt items
    // (description starts with "Belanja [Store] -")
    const receiptTransactions = await prisma.transaction.findMany({
        where: {
            description: {
                contains: 'Belanja'
            }
        },
        orderBy: {
            date: 'asc'
        }
    });

    // Group by date + store name
    const groups = new Map();

    for (const tx of receiptTransactions) {
        const match = tx.description.match(/Belanja (.+?) - (.+)/);
        if (match) {
            const storeName = match[1];
            const itemName = match[2];
            const key = `${tx.date.toISOString()}-${storeName}-${tx.userId}`;

            if (!groups.has(key)) {
                groups.set(key, []);
            }

            groups.get(key).push({
                transaction: tx,
                itemName
            });
        }
    }

    console.log(`Found ${groups.size} receipt groups to migrate`);

    // For each group, create 1 parent transaction + items
    for (const [key, items] of groups.entries()) {
        if (items.length < 2) continue; // Skip single transactions

        const first = items[0].transaction;
        const totalAmount = items.reduce((sum, i) => sum + i.transaction.amount, 0);

        console.log(`Migrating group: ${key}, ${items.length} items, total: ${totalAmount}`);

        // Create parent transaction
        const parent = await prisma.transaction.create({
            data: {
                userId: first.userId,
                walletId: first.walletId,
                categoryId: first.categoryId,
                description: `Belanja ${items[0].transaction.description.split(' - ')[0].replace('Belanja ', '')}`,
                amount: totalAmount,
                date: first.date,
                type: 'EXPENSE',
                // Create items
                items: {
                    create: items.map((item, index) => ({
                        name: item.itemName,
                        quantity: 1,
                        unitPrice: item.transaction.amount,
                        totalPrice: item.transaction.amount,
                        order: index
                    }))
                }
            }
        });

        // Delete old transactions
        await prisma.transaction.deleteMany({
            where: {
                id: {
                    in: items.map(i => i.transaction.id)
                }
            }
        });

        console.log(`Created parent transaction ${parent.id}, deleted ${items.length} old transactions`);
    }

    console.log('Migration complete!');
}

// Run: ts-node scripts/migrate-receipt-transactions.ts
migrateReceiptTransactions();
```

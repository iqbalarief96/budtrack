Task 2.1: Create GET Transaction Items API Route
-------------------------------------------------
File: app/api/transactions/[id]/items/route.ts

Function: GET - Fetch all items for a transaction

Implementation:
import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const transactionId = params.id

    // Verify transaction belongs to user
    const transaction = await prisma.transaction.findFirst({
      where: {
        id: transactionId,
        userId: session.user.id
      }
    })

    if (!transaction) {
      return NextResponse.json(
        { error: 'Transaction not found' },
        { status: 404 }
      )
    }

    // Fetch items ordered by 'order' field
    const items = await prisma.transactionItem.findMany({
      where: { transactionId },
      orderBy: { order: 'asc' }
    })

    return NextResponse.json({ items })
  } catch (error) {
    console.error('Error fetching transaction items:', error)
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    )
  }
}

Verification: API returns transaction items


Task 2.2: Create POST Transaction Item API Route
-------------------------------------------------
File: app/api/transactions/[id]/items/route.ts (POST method)

Function: POST - Add new item to transaction

Implementation:
export async function POST(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const transactionId = params.id
    const body = await req.json()
    const { name, quantity, unitPrice, category, notes } = body

    // Validate required fields
    if (!name || !unitPrice) {
      return NextResponse.json(
        { error: 'Name and unitPrice are required' },
        { status: 400 }
      )
    }

    // Verify transaction belongs to user
    const transaction = await prisma.transaction.findFirst({
      where: {
        id: transactionId,
        userId: session.user.id
      },
      include: {
        items: true
      }
    })

    if (!transaction) {
      return NextResponse.json(
        { error: 'Transaction not found' },
        { status: 404 }
      )
    }

    // Calculate total price
    const qty = quantity || 1
    const totalPrice = qty * unitPrice

    // Get next order number
    const maxOrder = transaction.items.length > 0
      ? Math.max(...transaction.items.map(i => i.order))
      : -1

    // Create transaction item
    const item = await prisma.transactionItem.create({
      data: {
        transactionId,
        name,
        quantity: qty,
        unitPrice,
        totalPrice,
        category,
        notes,
        order: maxOrder + 1
      }
    })

    // IMPORTANT: Update parent transaction amount
    const totalItemsAmount = await prisma.transactionItem.aggregate({
      where: { transactionId },
      _sum: { totalPrice: true }
    })

    await prisma.transaction.update({
      where: { id: transactionId },
      data: { amount: totalItemsAmount._sum.totalPrice || 0 }
    })

    return NextResponse.json({ item }, { status: 201 })
  } catch (error) {
    console.error('Error creating transaction item:', error)
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    )
  }
}

Verification: Item created and transaction amount updated

  req: Request,
  { params }: { params: { transactionId: string; itemId: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { transactionId, itemId } = params
    const body = await req.json()

    if (!transaction) {
      return NextResponse.json(
        { error: 'Transaction not found' },

    // Update parent transaction amount

      where: { transactionId },

    await prisma.transaction.update({
      where: { id: transactionId },
      data: { amount: totalItemsAmount._sum.totalPrice || 0 }
    })

    return NextResponse.json({ item })
  } catch (error) {
    console.error('Error updating transaction item:', error)

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function POST(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const transactionId = params.id
    const body = await req.json()
    const { items } = body // Array of items

    if (!Array.isArray(items) || items.length === 0) {
      return NextResponse.json(
        { error: 'Items array is required' },
        { status: 400 }
      )
    }

    // Verify transaction belongs to user
    const transaction = await prisma.transaction.findFirst({
      where: {
        id: transactionId,
        userId: session.user.id
      }
    })

    if (!transaction) {
      return NextResponse.json(
        { error: 'Transaction not found' },
        { status: 404 }
      )
    }

    // Create items with proper order
    const createdItems = await Promise.all(
      items.map((item, index) => {
        const qty = item.quantity || 1
        const totalPrice = qty * item.unitPrice

        return prisma.transactionItem.create({
          data: {
            transactionId,
            name: item.name,
            quantity: qty,
            unitPrice: item.unitPrice,
            totalPrice,
            category: item.category,
            notes: item.notes,
            order: index
          }
        })
      })
    )

    // Update parent transaction amount
    const totalItemsAmount = await prisma.transactionItem.aggregate({
      where: { transactionId },
      _sum: { totalPrice: true }
    })

    await prisma.transaction.update({
      where: { id: transactionId },
      data: { amount: totalItemsAmount._sum.totalPrice || 0 }
    })

    return NextResponse.json({ items: createdItems }, { status: 201 })
  } catch (error) {
    console.error('Error bulk creating transaction items:', error)
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    )
  }
}

Verification: Multiple items created at once

'use client'

import { useState } from 'react'
import { ChevronDown, ChevronRight, Plus, Pencil, Trash } from 'lucide-react'
import { Button } from '@/components/ui/button'
import { formatCurrency } from '@/lib/utils'

interface TransactionItem {
  id: string
  name: string
  quantity: number
  unitPrice: number
  totalPrice: number
  category?: string
  notes?: string
}

interface TransactionItemsListProps {
  transactionId: string
  items: TransactionItem[]
  onItemsChange: () => void
}

export default function TransactionItemsList({
  transactionId,
  items,
  onItemsChange
}: TransactionItemsListProps) {
  const [isExpanded, setIsExpanded] = useState(false)
  const [isLoading, setIsLoading] = useState(false)

  const toggleExpand = () => {
    setIsExpanded(!isExpanded)
  }

  const handleDeleteItem = async (itemId: string) => {
    if (!confirm('Delete this item?')) return

    setIsLoading(true)
    try {
      const response = await fetch(
        `/api/transactions/${transactionId}/items/${itemId}`,
        { method: 'DELETE' }
      )

      if (response.ok) {
        onItemsChange()
      }
    } catch (error) {
      console.error('Error deleting item:', error)
    } finally {
      setIsLoading(false)
    }
  }

  if (items.length === 0) return null

  return (
    <div className="mt-2 border-t pt-2">
      <button
        onClick={toggleExpand}
        className="flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground"
      >
        {isExpanded ? (
          <ChevronDown className="h-4 w-4" />
        ) : (
          <ChevronRight className="h-4 w-4" />
        )}
        <span>{items.length} items</span>
      </button>

      {isExpanded && (
        <div className="mt-2 space-y-2 ml-6">
          {items.map((item) => (
            <div
              key={item.id}
              className="flex items-start justify-between p-2 rounded-lg bg-muted/50 hover:bg-muted"
            >
              <div className="flex-1">
                <div className="font-medium">{item.name}</div>
                <div className="text-sm text-muted-foreground">
                  {item.quantity}x @ {formatCurrency(item.unitPrice)}
                </div>
                {item.notes && (
                  <div className="text-xs text-muted-foreground mt-1">
                    {item.notes}
                  </div>
                )}
              </div>
              <div className="flex items-center gap-2">
                <span className="font-semibold">
                  {formatCurrency(item.totalPrice)}
                </span>
                <Button
                  size="sm"
                  variant="ghost"
                  onClick={() => handleDeleteItem(item.id)}
                  disabled={isLoading}
                >
                  <Trash className="h-4 w-4" />
                </Button>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  )
}

Task 3.2: Create AddTransactionItemDialog Component

Purpose: Dialog for adding new item to transaction

Implementation:
'use client'

import { useState } from 'react'
import { Plus } from 'lucide-react'
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from '@/components/ui/dialog'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import { Textarea } from '@/components/ui/textarea'

interface AddTransactionItemDialogProps {
  transactionId: string
  onItemAdded: () => void
}

export default function AddTransactionItemDialog({
  transactionId,
  onItemAdded
}: AddTransactionItemDialogProps) {
  const [open, setOpen] = useState(false)
  const [isLoading, setIsLoading] = useState(false)
  const [formData, setFormData] = useState({
    name: '',
    quantity: 1,
    unitPrice: 0,
    category: '',
    notes: ''
  })

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setIsLoading(true)

    try {
      const response = await fetch(
        `/api/transactions/${transactionId}/items`,
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(formData)
        }
      )

      if (response.ok) {
        setOpen(false)
        setFormData({
          name: '',
          quantity: 1,
          unitPrice: 0,
          category: '',
          notes: ''
        })
        onItemAdded()
      }
    } catch (error) {
      console.error('Error adding item:', error)
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>
        <Button variant="outline" size="sm">
          <Plus className="h-4 w-4 mr-2" />
          Add Item
        </Button>
      </DialogTrigger>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Add Transaction Item</DialogTitle>
        </DialogHeader>
        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <Label htmlFor="name">Item Name</Label>
            <Input
              id="name"
              value={formData.name}
              onChange={(e) =>
                setFormData({ ...formData, name: e.target.value })
              }
              placeholder="e.g., Susu Ultra 1L"
              required
            />
          </div>
          <div className="grid grid-cols-2 gap-4">
            <div>
              <Label htmlFor="quantity">Quantity</Label>
              <Input
                id="quantity"
                type="number"
                step="0.01"
                min="0"
                value={formData.quantity}
                onChange={(e) =>
                  setFormData({
                    ...formData,
                    quantity: parseFloat(e.target.value)
                  })
                }
                required
              />
            </div>
            <div>
              <Label htmlFor="unitPrice">Unit Price</Label>
              <Input
                id="unitPrice"
                type="number"
                step="0.01"
                min="0"
                value={formData.unitPrice}
                onChange={(e) =>
                  setFormData({
                    ...formData,
                    unitPrice: parseFloat(e.target.value)
                  })
                }
                required
              />
            </div>
          </div>
          <div>
            <Label htmlFor="category">Category (optional)</Label>
            <Input
              id="category"
              value={formData.category}
              onChange={(e) =>
                setFormData({ ...formData, category: e.target.value })
              }
              placeholder="e.g., Groceries, Dairy"
            />
          </div>
          <div>
            <Label htmlFor="notes">Notes (optional)</Label>
            <Textarea
              id="notes"
              value={formData.notes}
              onChange={(e) =>
                setFormData({ ...formData, notes: e.target.value })
              }
              placeholder="Any additional notes..."
            />
          </div>
          <div className="flex justify-between items-center">
            <div className="text-sm">
              Total: <strong>Rp {(formData.quantity * formData.unitPrice).toLocaleString()}</strong>
            </div>
            <Button type="submit" disabled={isLoading}>
              {isLoading ? 'Adding...' : 'Add Item'}
            </Button>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  )
}
